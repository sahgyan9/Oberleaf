\section{Main Result}

\begin{theorem}\label{thm:main}
For every $n \ge 1$,
\begin{equation}
  \sum_{k=1}^{n} k = \frac{n(n+1)}{2}.
\end{equation}
\end{theorem}

\begin{proof}
By induction on $n$. The case $n = 1$ is immediate, and adding $n + 1$ to
both sides gives the statement for $n + 1$.
\end{proof}
